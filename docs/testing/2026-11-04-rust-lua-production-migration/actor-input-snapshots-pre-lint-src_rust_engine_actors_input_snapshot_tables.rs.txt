//! The existing immutable dense Boolean-table array schema.
use super::*;

pub(super) fn capture_array(
    stack: &mut NativeStack<'_, '_>,
    root: TableId,
    object: u32,
    first: u32,
    actors: &FxHashMap<TableId, usize>,
    reserved: &[TableId],
    ids: &mut FxHashMap<TableId, u32>,
    bindings: &mut program::Bindings,
) -> Result<(), Error> {
    let count = raw_len(stack, root)?;
    if count == 0 || count > 4096 {
        return Err(stack.error(b"unsupported native state array size"));
    }
    let mut children = vec![None; count];
    for (key, value) in raw_entries(stack, root)? {
        let Value::Number(index) = key else {
            return Err(stack.error(b"native state array needs dense numeric keys"));
        };
        if !(index >= 1. && index <= count as f64 && index.fract() == 0.) {
            return Err(stack.error(b"native state array needs dense numeric keys"));
        }
        let Value::Table(child) = value else {
            return Err(stack.error(b"native state array needs table references"));
        };
        children[index as usize - 1] = Some(child);
    }
    let mut references = Vec::with_capacity(count);
    for child in children {
        let child = child.ok_or_else(|| stack.error(b"native state array has a hole"))?;
        if actors.contains_key(&child)
            || reserved.contains(&child)
            || metatable(stack, child)?.is_some()
        {
            return Err(stack.error(b"unsupported native state array target"));
        }
        let id = if let Some(&id) = ids.get(&child) {
            if !bindings.boolean_tables.iter().any(|map| map.object == id) {
                return Err(stack.error(b"native state table ownership overlaps"));
            }
            id
        } else {
            let id = u32::try_from(ids.len())
                .ok()
                .and_then(|offset| first.checked_add(offset))
                .ok_or_else(|| stack.error(b"too many native state tables"))?;
            let mut entries = Vec::new();
            for (key, value) in raw_entries(stack, child)? {
                let key = byte_string(stack, key)?
                    .ok_or_else(|| stack.error(b"native Boolean state needs byte keys"))?;
                let Value::Boolean(value) = value else {
                    return Err(stack.error(b"native Boolean state needs Boolean values"));
                };
                entries.push((key, value));
            }
            entries.sort();
            bindings.boolean_tables.push(program::BooleanTable {
                object: id,
                entries,
            });
            ids.insert(child, id);
            id
        };
        references.push(id);
    }
    bindings
        .boolean_table_arrays
        .push(program::BooleanTableArray {
            object,
            tables: references,
        });
    Ok(())
}
