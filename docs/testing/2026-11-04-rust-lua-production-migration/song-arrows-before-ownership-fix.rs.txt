//! Existing song ArrowEffects bindings and their post-health live cache phase.
use super::*;
use std::sync::atomic::Ordering;
mod math;

#[derive(Clone, Copy)]
struct Cache {
    beat_x: [f32; LUA_PLAYERS],
    beat_y: [f32; LUA_PLAYERS],
    tipsy: [[f32; 10]; LUA_PLAYERS],
    tan_tipsy: [[f32; 10]; LUA_PLAYERS],
    bounds: [[[f32; 2]; 10]; LUA_PLAYERS],
    invert: [[f32; 10]; LUA_PLAYERS],
}
impl Default for Cache {
    fn default() -> Self {
        Self {
            beat_x: [0.; LUA_PLAYERS],
            beat_y: [0.; LUA_PLAYERS],
            tipsy: [[0.; 10]; LUA_PLAYERS],
            tan_tipsy: [[0.; 10]; LUA_PLAYERS],
            bounds: [[[0.; 2]; 10]; LUA_PLAYERS],
            invert: [[0.; 10]; LUA_PLAYERS],
        }
    }
}
#[derive(Default)]
struct State {
    cache: Mutex<Cache>,
    #[cfg(feature = "test-support")]
    clocks: Mutex<Option<crate::clock_inputs::ArrowFrameClocks>>,
}
struct Data {
    state: Arc<State>,
    clock: Arc<Clock>,
    enabled: [bool; LUA_PLAYERS],
    height: f32,
    rate: f32,
    timing: [Option<deadsync_rules::timing::TimingData>; LUA_PLAYERS],
    #[cfg(feature = "test-support")]
    inputs: Arc<Mutex<super::host_clocks::Inputs>>,
}
#[derive(Clone)]
struct Binding {
    data: Arc<Data>,
    name: &'static str,
}
pub(super) struct Arrows {
    root: Root,
    #[cfg(feature = "test-support")]
    state: Arc<State>,
}
pub(super) fn install(
    vm: &mut Vm,
    catalog: Value,
    clock: Arc<Clock>,
    context: &SongLuaCompileContext,
) -> Result<Arrows, Error> {
    let batch = vm.try_new_table()?;
    let root = vm.pin(Value::Table(batch))?;
    let state = Arc::new(State::default());
    let data = Arc::new(Data {
        state: state.clone(),
        clock,
        enabled: context.players.each_ref().map(|p| p.enabled),
        height: context.screen_height.max(1.),
        rate: crate::song_music_rate(context),
        timing: context.player_timing.clone(),
        #[cfg(feature = "test-support")]
        inputs: Arc::new(Mutex::new(super::host_clocks::Inputs::new(context))),
    });
    let result = (|| {
        let public = vm.try_new_table()?;
        vm.raw_table_set(batch, Value::Number(3.), Value::Table(public))?;
        let globals = vm.with_native_stack(|s| Ok(s.value(-10002).unwrap()))?;
        let tonumber = vm.get_global(b"tonumber")?;
        for name in [
            "GetYOffset",
            "GetYPos",
            "GetXPos",
            "Update",
            "GetZPos",
            "GetRotationX",
            "GetRotationY",
            "GetRotationZ",
            "GetZoom",
            "GetAlpha",
            "GetGlow",
        ] {
            let f = vm.new_host_function_with_data(
                callback,
                &[globals, catalog, Value::Table(batch), tonumber],
                Binding {
                    data: data.clone(),
                    name,
                },
            )?;
            if name == "Update" {
                vm.raw_table_set(batch, Value::Number(2.), f)?;
            }
            let pinned = vm.pin(f)?;
            let published = super::put(vm, public, name, f);
            vm.unpin(pinned)?;
            published?;
        }
        vm.set_global(b"ArrowEffects", Value::Table(public))?;
        // Publication owns the public table now. Retain only the private cache
        // and original updater, so clearing the Lua global can collect the API.
        vm.raw_table_set(batch, Value::Number(3.), Value::Nil)
    })();
    if let Err(e) = result {
        vm.unpin(root)?;
        return Err(e);
    }
    Ok(Arrows {
        root,
        #[cfg(feature = "test-support")]
        state,
    })
}
impl Arrows {
    fn update(&self, vm: &mut Vm) -> Result<(), Error> {
        let Value::Table(batch) = vm.rooted(self.root)? else {
            unreachable!()
        };
        let f = vm.raw_table_get(batch, Value::Number(2.))?;
        vm.call(f, &[]).map(|_| ())
    }
}
impl RustSongRuntime {
    pub fn update_arrow_effects(&mut self) -> Result<(), Error> {
        self.arrows.update(&mut self.vm)
    }
    #[cfg(feature = "test-support")]
    pub fn set_arrow_clock_inputs(
        &mut self,
        tipsy: [Option<f64>; LUA_PLAYERS],
        tan_tipsy: [Option<f64>; LUA_PLAYERS],
    ) {
        *self.arrows.state.clocks.lock().unwrap() =
            Some(crate::clock_inputs::ArrowFrameClocks { tipsy, tan_tipsy });
    }
}
fn callback(call: &mut HostCall<'_>) -> Result<(), Error> {
    let b = call.function_data::<Binding>().unwrap().clone();
    let args = call.arguments().to_vec();
    let result = match b.name {
        "Update" => return update(call, &b.data),
        "GetYOffset" => y_offset(call, &b.data, &args)?,
        "GetYPos" => math::y_pos(call, &b.data, &args)?,
        "GetXPos" => math::x_pos(call, &b.data, &args)?,
        "GetZoom" => math::zoom(call, &b.data, &args)?,
        "GetAlpha" => math::alpha(call, &b.data, &args)?,
        "GetRotationZ" => math::rotation_z(call, &b.data, &args)?,
        "GetRotationX" => {
            let _ = player_options(call, &args)?;
            crate::host::arrow_effects_rotation_x(loose(call, &args, 2, 1.) as i32)
        }
        "GetRotationY" | "GetZPos" | "GetGlow" => {
            let _ = player_options(call, &args)?;
            0.
        }
        _ => unreachable!(),
    };
    call.push(number(result))
}
fn table(call: &HostCall<'_>, v: Value) -> Result<TableId, Error> {
    match v {
        Value::Table(t) => Ok(t),
        _ => Err(call.error(b"ArrowEffects table expected")),
    }
}
fn field(call: &mut HostCall<'_>, owner: Value, key: &str, raw: bool) -> Result<Value, Error> {
    call.with_roots(&[owner], |call| {
        let key = call.try_intern(key.as_bytes())?;
        if raw {
            call.raw_table_get(table(call, owner)?, key)
        } else {
            call.table_get(owner, key)
        }
    })
}
fn raw(call: &mut HostCall<'_>, owner: TableId, key: &str) -> Result<Value, Error> {
    field(call, Value::Table(owner), key, true)
}
fn numeric(call: &mut HostCall<'_>, v: Value) -> Result<f64, Error> {
    coerce_number(call, v)?.ok_or_else(|| call.error(b"number expected"))
}
fn coerce_number(call: &mut HostCall<'_>, v: Value) -> Result<Option<f64>, Error> {
    match v {
        Value::Number(n) => Ok(Some(n)),
        Value::String(_) => {
            let value = call
                .call(call.captures()[3], &[v])?
                .first()
                .copied()
                .unwrap_or(Value::Nil);
            Ok(if let Value::Number(n) = value {
                Some(n)
            } else {
                None
            })
        }
        _ => Ok(None),
    }
}
fn strict(call: &mut HostCall<'_>, args: &[Value], i: usize, default: f32) -> Result<f32, Error> {
    let v = args.get(i).copied().unwrap_or(Value::Nil);
    if v == Value::Nil {
        return Ok(default);
    }
    coerce_number(call, v)?
        .map(|n| n as f32)
        .ok_or_else(|| call.error(b"expected numeric ArrowEffects argument"))
}
fn read_float(call: &HostCall<'_>, v: Value) -> Option<f32> {
    match v {
        Value::Number(n) => {
            let n = n as f32;
            n.is_finite().then_some(n)
        }
        Value::String(id) => std::str::from_utf8(call.string_bytes(id)?)
            .ok()?
            .trim()
            .parse()
            .ok(),
        _ => None,
    }
}
fn loose(call: &HostCall<'_>, args: &[Value], i: usize, default: f32) -> f32 {
    args.get(i)
        .copied()
        .and_then(|v| read_float(call, v))
        .unwrap_or(default)
}
fn n(call: &mut HostCall<'_>, state: Option<TableId>, key: &str) -> Result<f32, Error> {
    let Some(t) = state else { return Ok(0.) };
    let v = raw(call, t, key)?;
    if v == Value::Nil {
        Ok(0.)
    } else {
        numeric(call, v).map(|n| n as f32)
    }
}
fn flag(call: &mut HostCall<'_>, state: Option<TableId>, key: &str) -> Result<bool, Error> {
    let Some(t) = state else { return Ok(false) };
    Ok(!matches!(
        raw(call, t, key)?,
        Value::Nil | Value::Boolean(false)
    ))
}
fn optional_table(call: &HostCall<'_>, v: Value) -> Result<Option<TableId>, Error> {
    if v == Value::Nil {
        Ok(None)
    } else {
        table(call, v).map(Some)
    }
}
fn invoke(
    call: &mut HostCall<'_>,
    owner: Value,
    name: &str,
    args: &[Value],
) -> Result<Value, Error> {
    call.with_roots(&[owner], |call| {
        let f = field(call, owner, name, false)?;
        if !matches!(f, Value::Function(_) | Value::Native(_)) {
            return Err(call.error(b"ArrowEffects method expected"));
        }
        let mut values = vec![owner];
        values.extend_from_slice(args);
        Ok(call
            .call(f, &values)?
            .first()
            .copied()
            .unwrap_or(Value::Nil))
    })
}
fn player_options(call: &mut HostCall<'_>, args: &[Value]) -> Result<Option<TableId>, Error> {
    let Some(&Value::Table(player)) = args.first() else {
        return Ok(None);
    };
    let f = field(call, Value::Table(player), "GetCurrentPlayerOptions", false)?;
    if f == Value::Nil {
        return Ok(None);
    };
    if !matches!(f, Value::Function(_) | Value::Native(_)) {
        return Err(call.error(b"GetCurrentPlayerOptions must be a function"));
    }
    let options = call
        .call(f, &[Value::Table(player)])?
        .first()
        .copied()
        .unwrap_or(Value::Nil);
    let options = table(call, options)?;
    let batch = table(call, call.captures()[2])?;
    call.raw_table_set(batch, Value::Number(1.), Value::Table(options))?;
    Ok(Some(options))
}
fn with_options<T>(
    call: &mut HostCall<'_>,
    args: &[Value],
    f: impl FnOnce(&mut HostCall<'_>, Option<TableId>, Option<TableId>) -> Result<T, Error>,
) -> Result<T, Error> {
    let options = player_options(call, args)?;
    let state = match options {
        Some(t) => {
            let v = raw(call, t, "__songlua_player_option_state")?;
            optional_table(call, v)?
        }
        None => None,
    };
    call.with_roots(
        &[
            options.map_or(Value::Nil, Value::Table),
            state.map_or(Value::Nil, Value::Table),
        ],
        |call| f(call, options, state),
    )
}
fn current_options(call: &mut HostCall<'_>, player: usize) -> Result<TableId, Error> {
    let catalog = table(call, call.captures()[1])?;
    let players = raw(call, catalog, "player_options")?;
    let players = table(call, players)?;
    let levels = call.raw_table_get(players, Value::Number((player + 1) as f64))?;
    let levels = table(call, levels)?;
    let t = call.raw_table_get(levels, Value::Number(4.))?;
    table(call, t)
}
fn player_for(call: &mut HostCall<'_>, options: Option<TableId>) -> Result<usize, Error> {
    if let Some(t) = options {
        for p in 0..LUA_PLAYERS {
            if current_options(call, p)? == t {
                return Ok(p);
            }
        }
    }
    Ok(0)
}
fn player_index(call: &HostCall<'_>, value: Value) -> Option<usize> {
    match value {
        Value::Number(n) if n >= 0. && n < LUA_PLAYERS as f64 && n.fract() == 0. => {
            Some(n as usize)
        }
        Value::String(id) => {
            let bytes = call.string_bytes(id)?;
            if bytes.eq_ignore_ascii_case(b"P1") || bytes.eq_ignore_ascii_case(b"PlayerNumber_P1") {
                Some(0)
            } else if bytes.eq_ignore_ascii_case(b"P2")
                || bytes.eq_ignore_ascii_case(b"PlayerNumber_P2")
            {
                Some(1)
            } else {
                None
            }
        }
        _ => None,
    }
}
fn runtime(call: &mut HostCall<'_>, is_raw: bool) -> Result<TableId, Error> {
    let t = field(call, call.captures()[0], SONG_LUA_RUNTIME_KEY, is_raw)?;
    table(call, t)
}
fn position_beat(
    call: &mut HostCall<'_>,
    runtime: TableId,
    positions: Option<TableId>,
    player: usize,
) -> Result<f32, Error> {
    let position = if let Some(t) = positions {
        let v = call.raw_table_get(t, Value::Number((player + 1) as f64))?;
        table(call, v)?
    } else {
        runtime
    };
    let v = raw(call, position, BEAT)?;
    numeric(call, v).map(|n| n as f32)
}
fn timer(call: &mut HostCall<'_>, state: Option<TableId>) -> Result<u8, Error> {
    let v = if let Some(t) = state {
        raw(call, t, "modtimersetting")?
    } else {
        Value::Nil
    };
    if v == Value::Nil {
        return Ok(3);
    };
    coerce_number(call, v)?
        .filter(|n| n.is_finite() && *n >= i64::MIN as f64 && *n < -(i64::MIN as f64))
        .and_then(|n| u8::try_from(n as i64).ok())
        .ok_or_else(|| call.error(b"invalid mod timer setting"))
}
fn time(
    call: &mut HostCall<'_>,
    state: Option<TableId>,
    raw_uptime: Option<f64>,
    is_raw: bool,
) -> Result<f32, Error> {
    clock_values(call, state, raw_uptime, is_raw).map(|(_, t)| t)
}
fn alpha_clock(call: &mut HostCall<'_>, state: Option<TableId>) -> Result<(f32, f32), Error> {
    clock_values(call, state, None, false)
}
fn clock_values(
    call: &mut HostCall<'_>,
    state: Option<TableId>,
    raw_uptime: Option<f64>,
    is_raw: bool,
) -> Result<(f32, f32), Error> {
    let t = runtime(call, is_raw)?;
    call.with_roots(&[Value::Table(t)], |call| {
        let v = field(call, Value::Table(t), BEAT, is_raw)?;
        let beat = numeric(call, v)? as f32;
        let v = field(call, Value::Table(t), SECONDS, is_raw)?;
        let seconds = numeric(call, v)? as f32;
        let v = field(call, Value::Table(t), "__songlua_uptime_seconds", is_raw)?;
        let uptime = if v == Value::Nil {
            None
        } else {
            Some(numeric(call, v)?)
        };
        let uptime = raw_uptime.or(uptime).unwrap_or(f64::from(seconds));
        let base = match timer(call, state)? {
            1 => f64::from(beat),
            2 => f64::from(seconds),
            _ => uptime,
        };
        Ok((
            beat,
            ((base + f64::from(n(call, state, "modtimeroffset")?))
                * (1. + f64::from(n(call, state, "modtimermult")?))) as f32,
        ))
    })
}
fn query_time(
    call: &mut HostCall<'_>,
    b: &Data,
    state: Option<TableId>,
    _domain: &str,
) -> Result<f32, Error> {
    #[cfg(feature = "test-support")]
    let raw = if matches!(timer(call, state)?, 1 | 2) {
        None
    } else {
        super::host_clocks::Inputs::next(&b.inputs, call, _domain)?
    };
    #[cfg(not(feature = "test-support"))]
    let raw = {
        let _ = b;
        None
    };
    time(call, state, raw, true)
}
fn style_name(call: &mut HostCall<'_>) -> String {
    let selected = (|| {
        let game = field(call, call.captures()[0], "GAMESTATE", false)?;
        let t = field(call, game, "__songlua_current_style", true)?;
        let t = table(call, t)?;
        call.raw_table_get(t, Value::Number(1.))
    })();
    let selected = selected.or_else(|_| {
        let g = field(call, call.captures()[0], "GAMESTATE", false)?;
        invoke(call, g, "GetCurrentStyle", &[])
    });
    let Ok(Value::Table(t)) = selected else {
        return "single".into();
    };
    call.with_roots(&[Value::Table(t)], |call| {
        let key = raw(call, t, "__songlua_style_key");
        if let Ok(v) = key
            && v != Value::Nil
            && let Some(id) = call.coerce_string(v)?
            && let Ok(key) = std::str::from_utf8(call.string_bytes(id).unwrap())
        {
            return Ok(key.to_owned());
        }
        let v = invoke(call, Value::Table(t), "GetName", &[])?;
        match v {
            Value::String(id) => {
                Ok(String::from_utf8_lossy(call.string_bytes(id).unwrap()).into_owned())
            }
            _ => Ok("single".into()),
        }
    })
    .unwrap_or_else(|_| "single".into())
}
fn speed_value(
    call: &mut HostCall<'_>,
    options: TableId,
    name: &str,
) -> Result<Option<f32>, Error> {
    let f = field(call, Value::Table(options), name, false)?;
    if f == Value::Nil {
        return Ok(None);
    };
    if !matches!(f, Value::Function(_) | Value::Native(_)) {
        return Err(call.error(b"speed modifier must be a function"));
    }
    let v = call
        .call(f, &[Value::Table(options)])?
        .first()
        .copied()
        .unwrap_or(Value::Nil);
    Ok(read_float(call, v).filter(|n| n.is_finite() && *n > 0.))
}
fn speed_multiplier(call: &mut HostCall<'_>, args: &[Value]) -> Result<f32, Error> {
    let Some(options) = player_options(call, args)? else {
        return Ok(1.);
    };
    call.with_roots(&[Value::Table(options)], |call| {
        if let Some(x) = speed_value(call, options, "XMod")? {
            return Ok(x);
        }
        let v = field(
            call,
            Value::Table(options),
            "__songlua_reference_bpm",
            false,
        )?;
        let reference = if v == Value::Nil {
            None
        } else {
            Some(numeric(call, v)? as f32)
        }
        .filter(|n| n.is_finite() && *n > 0.)
        .unwrap_or(1.);
        for name in ["CMod", "MMod", "AMod"] {
            if let Some(v) = speed_value(call, options, name)? {
                return Ok(v / reference);
            }
        }
        Ok(1.)
    })
}
fn reverse_percent(call: &mut HostCall<'_>, args: &[Value]) -> Result<f32, Error> {
    let Some(options) = player_options(call, args)? else {
        return Ok(0.);
    };
    call.with_roots(&[Value::Table(options)], |call| {
        let f = field(
            call,
            Value::Table(options),
            "GetReversePercentForColumn",
            false,
        )?;
        if f == Value::Nil {
            return Ok(0.);
        }
        if !matches!(f, Value::Function(_) | Value::Native(_)) {
            return Err(call.error(b"reverse modifier must be a function"));
        }
        let column = args
            .get(1)
            .copied()
            .and_then(|v| read_float(call, v))
            .map(|n| n - 1.)
            .unwrap_or(0.);
        let v = call
            .call(f, &[Value::Table(options), number(column)])?
            .first()
            .copied()
            .unwrap_or(Value::Nil);
        Ok(read_float(call, v).unwrap_or(0.))
    })
}
fn wave_offset(
    call: &mut HostCall<'_>,
    args: &[Value],
    mut y: f32,
    speed: f32,
) -> Result<f32, Error> {
    if y >= 0. {
        with_options(call, args, |call, _, state| {
            let wave = n(call, state, "wave")?;
            if wave != 0. {
                let period = n(call, state, "waveperiod")?;
                y += wave * 20. * (y / (period * 38. + 38.)).sin();
            }
            Ok(())
        })?;
    }
    Ok(y * speed)
}
fn y_offset(call: &mut HostCall<'_>, b: &Data, args: &[Value]) -> Result<f32, Error> {
    let speed = speed_multiplier(call, args)?;
    let note = loose(call, args, 2, 0.);
    if b.timing.iter().all(Option::is_none) {
        return wave_offset(call, args, 64. * note, speed);
    }
    let player = if let Some(&Value::Table(t)) = args.first() {
        let v = invoke(call, Value::Table(t), "GetPlayerNumber", &[])?;
        player_index(call, v).unwrap_or(0)
    } else {
        0
    };
    let Some(timing) = b.timing.get(player).and_then(Option::as_ref) else {
        return wave_offset(call, args, 64. * note, speed);
    };
    let t = runtime(call, false)?;
    let (beat, seconds) = call.with_roots(&[Value::Table(t)], |call| {
        let v = field(call, Value::Table(t), BEAT, false)?;
        let beat = numeric(call, v)? as f32;
        let v = field(call, Value::Table(t), SECONDS, false)?;
        let seconds = numeric(call, v)?;
        let seconds = if b.clock.live.load(Ordering::Relaxed) {
            seconds as f32
        } else {
            (seconds * f64::from(b.rate)) as f32
                + b.clock
                    .timing
                    .as_ref()
                    .map_or(0., |t| t.get_time_for_beat_exact(0.))
        };
        Ok((beat, seconds))
    })?;
    let options = player_options(call, args)?;
    if let Some(options) = options {
        call.with_roots(&[Value::Table(options)], |call| {
            if let Some(cmod) = speed_value(call, options, "CMod")? {
                return wave_offset(
                    call,
                    args,
                    (timing.get_time_for_beat(note) - seconds) * cmod / 60. / b.rate * 64.,
                    1.,
                );
            }
            let speed = if speed_value(call, options, "XMod")?.is_some() {
                speed
            } else {
                speed / b.rate
            };
            wave_offset(
                call,
                args,
                (timing.get_displayed_beat(note) - timing.get_displayed_beat(beat))
                    * timing.get_speed_multiplier(beat, seconds)
                    * 64.,
                speed,
            )
        })
    } else {
        wave_offset(
            call,
            args,
            (timing.get_displayed_beat(note) - timing.get_displayed_beat(beat))
                * timing.get_speed_multiplier(beat, seconds)
                * 64.,
            1.,
        )
    }
}
// Preserve the recorded native f32 operation sequence, including nonfinite results.
#[allow(clippy::neg_multiply)]
fn beat_factor(beat: f32, offset: f32, multiplier: f32) -> f32 {
    let mut phase = (beat + 0.2 + offset) * (multiplier + 1.);
    let reverse = phase as i32 % 2 != 0;
    if phase < 0. {
        return 0.;
    }
    phase -= phase.trunc();
    phase += 1.;
    phase -= phase.trunc();
    if phase >= 0.5 {
        return 0.;
    }
    let mut factor = if phase < 0.2 {
        let v = phase / 0.2;
        v * v
    } else {
        let v = (phase - 0.2) * -1. / (0.5 - 0.2) + 1.;
        1. - (1. - v) * (1. - v)
    };
    if reverse {
        factor *= -1.
    }
    factor * 20.
}
#[allow(clippy::neg_multiply)] // Same native f32 sequence as beat_factor.
fn update(call: &mut HostCall<'_>, b: &Data) -> Result<(), Error> {
    let previous = *b.state.cache.lock().unwrap();
    let t = runtime(call, true)?;
    let v = raw(call, t, PLAYERS)?;
    let positions = optional_table(call, v)?;
    call.with_roots(
        &[Value::Table(t), positions.map_or(Value::Nil, Value::Table)],
        |call| {
            let mut next = previous;
            let style = crate::song_lua_style_info(&style_name(call));
            let two_sides = matches!(
                style.style_type,
                "StyleType_OnePlayerTwoSides" | "StyleType_TwoPlayersSharedSides"
            );
            let per_side = style.columns / if two_sides { 2 } else { 1 };
            let scale = |x: f32, low: f32, high: f32, to_low: f32, to_high: f32| {
                (x - low) * (to_high - to_low) / (high - low) + to_low
            };
            for player in 0..LUA_PLAYERS {
                if !b.enabled[player] {
                    continue;
                }
                let beat = position_beat(call, t, positions, player)?;
                let options = current_options(call, player)?;
                let v = raw(call, options, "__songlua_player_option_state")?;
                let state = optional_table(call, v)?;
                call.with_roots(
                    &[
                        Value::Table(options),
                        state.map_or(Value::Nil, Value::Table),
                    ],
                    |call| {
                        let mut phase = (beat + 0.2 + n(call, state, "beatoffset")?)
                            * (n(call, state, "beatmult")? + 1.);
                        let reverse = phase as i32 % 2 != 0;
                        let mut factor = 0.;
                        if phase >= 0. {
                            phase -= phase.trunc();
                            phase += 1.;
                            phase -= phase.trunc();
                            if phase < 0.5 {
                                factor = if phase < 0.2 {
                                    let v = phase / 0.2;
                                    v * v
                                } else {
                                    let v = (phase - 0.2) * -1. / (0.5 - 0.2) + 1.;
                                    1. - (1. - v) * (1. - v)
                                };
                                if reverse {
                                    factor *= -1.;
                                }
                                factor *= 20.;
                            }
                        }
                        next.beat_x[player] = factor;
                        next.beat_y[player] = beat_factor(
                            beat,
                            n(call, state, "beatyoffset")?,
                            n(call, state, "beatymult")?,
                        );
                        let batch = table(call, call.captures()[2])?;
                        let selected = call.raw_table_get(batch, Value::Number(1.))?;
                        let selected = if selected == Value::Nil {
                            options
                        } else {
                            table(call, selected)?
                        };
                        let selected_state = raw(call, selected, "__songlua_player_option_state")?;
                        let selected_state = optional_table(call, selected_state)?;
                        call.with_roots(
                            &[
                                Value::Table(selected),
                                selected_state.map_or(Value::Nil, Value::Table),
                            ],
                            |call| {
                                let cosecant = flag(call, selected_state, "cosecant")?;
                                let mut times = [0.; 2];
                                for (i, prefix) in ["tipsy", "tantipsy"].into_iter().enumerate() {
                                    if n(call, state, prefix)? == 0. {
                                        continue;
                                    }
                                    #[cfg(feature = "test-support")]
                                    let raw =
                                        b.state.clocks.lock().unwrap().as_ref().and_then(|c| {
                                            if i == 0 {
                                                c.tipsy[player]
                                            } else {
                                                c.tan_tipsy[player]
                                            }
                                        });
                                    #[cfg(not(feature = "test-support"))]
                                    let raw = None;
                                    times[i] = time(call, selected_state, raw, true)?;
                                }
                                for column in 0..style.columns {
                                    for (prefix, tan) in [("tipsy", false), ("tantipsy", true)] {
                                        let result = if n(call, state, prefix)? == 0. {
                                            0.
                                        } else {
                                            let angle = times[usize::from(tan)]
                                                * (n(call, state, &format!("{prefix}speed"))?
                                                    * 1.2
                                                    + 1.2)
                                                + column as f32
                                                    * (n(call, state, &format!("{prefix}offset"))?
                                                        * 1.8
                                                        + 1.8);
                                            let wave = if tan {
                                                if cosecant {
                                                    1. / angle.sin()
                                                } else {
                                                    angle.tan()
                                                }
                                            } else {
                                                angle.cos()
                                            };
                                            wave * (64. * 0.4)
                                        };
                                        if tan {
                                            next.tan_tipsy[player][column] = result
                                        } else {
                                            next.tipsy[player][column] = result
                                        }
                                    }
                                    let width = if style.columns > 4 { 2 } else { 3 };
                                    let range = &style.x_offsets[column.saturating_sub(width)
                                        ..=(column + width).min(style.columns - 1)];
                                    next.bounds[player][column] = [
                                        range.iter().copied().fold(f32::MAX, f32::min),
                                        range.iter().copied().fold(f32::MIN_POSITIVE, f32::max),
                                    ];
                                    let local = column % per_side;
                                    let (first, last) = if local <= (per_side - 1) / 2 {
                                        (0, (per_side - 1) / 2)
                                    } else if local >= per_side.div_ceil(2) {
                                        (per_side.div_ceil(2), per_side - 1)
                                    } else {
                                        (local / 2, local / 2)
                                    };
                                    let mirrored = if first == last {
                                        0
                                    } else {
                                        scale(
                                            local as f32,
                                            first as f32,
                                            last as f32,
                                            last as f32,
                                            first as f32,
                                        ) as usize
                                    };
                                    next.invert[player][column] = style.x_offsets
                                        [column / per_side * per_side + mirrored]
                                        - style.x_offsets[column];
                                }
                                Ok(())
                            },
                        )
                    },
                )?;
            }
            *b.state.cache.lock().unwrap() = next;
            #[cfg(feature = "test-support")]
            {
                *b.state.clocks.lock().unwrap() = None;
            }
            Ok(())
        },
    )
}
