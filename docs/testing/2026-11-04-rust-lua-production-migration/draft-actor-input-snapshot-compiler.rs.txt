// Draft only. The caller gates playback/capture/hooks and restores the complete
// idle native-stack window on every result. Original callbacks stay rooted in
// Input's private map; scene actors and updates have traced scene/plan owners.
fn compile_with_updates(
    stack: &mut NativeStack<'_, '_>, store: &Shared, inputs: &[Value],
    actors: &[TableId], states: &[SongLuaOverlayState],
    tracked: &[(SongLuaTrackedActorTarget, TableId, SongLuaOverlayState)],
    updates: &[(Value, usize)], reserved: &[TableId],
) -> Option<CompiledActorInputs> {
    let input_count = inputs.len();
    let callback_count = input_count + updates.len();
    let callbacks = || inputs.iter().chain(updates.iter().map(|(function, _)| function));
    let tables = actors.iter().copied().chain(tracked.iter().map(|(_, table, _)| *table)).collect::<Vec<_>>();
    let indices = tables.iter().copied().enumerate().map(|(index, table)| (table, index))
        .collect::<FxHashMap<_, _>>();
    if indices.len() != tables.len() || actors.len() != states.len() { return None; }
    u32::try_from(tables.len()).ok()?.checked_mul(4)?.checked_add(2)?;
    let mut bindings = super::super::input::bindings();
    let first = u32::try_from(tables.len()).ok()?.checked_add(2)?;
    let mut ids = FxHashMap::<TableId, u32>::default();
    let mut byte_fields = Vec::new();
    let mut pending = updates.iter().map(|(_, actor)| *actor).collect::<Vec<_>>();
    let mut captures = Vec::with_capacity(callback_count);
    for callback in callbacks() {
        if !matches!(callback, Value::Function(_)) { return None; }
        stack.push(*callback).ok()?;
        let function_index = stack.top();
        let mut upvalues = Vec::new();
        for number in 1.. {
            if stack.get_upvalue(function_index as isize, number).ok()?.is_none() { break; }
            let Value::Table(table) = stack.value(-1)? else { return None; };
            let object = if let Some(&index) = indices.get(&table) {
                pending.push(index); index as u32 + 2
            } else if reserved.contains(&table) {
                return None;
            } else if let Some(&object) = ids.get(&table) {
                object
            } else {
                if metatable(stack, table).ok()?.is_some() { return None; }
                let object = u32::try_from(ids.len()).ok().and_then(|offset| first.checked_add(offset))?;
                ids.insert(table, object);
                if raw_len(stack, table).ok()? > 0 {
                    tables::capture_array(stack, table, object, first, &indices, reserved, &mut ids, &mut bindings).ok()?;
                } else {
                    for (key, value) in raw_entries(stack, table).ok()? {
                        let name = byte_string(stack, key).ok()??;
                        let cell = match value {
                            Value::Boolean(value) => program::Cell::Boolean(value),
                            Value::Number(value) => program::Cell::Number(value),
                            Value::String(_) => {
                                let bytes = byte_string(stack, value).ok()??;
                                byte_fields.push(bindings.fields.len());
                                bindings.fields.push(Field { object, name: name.into_vec().into(),
                                    value: FieldValue::Value(Binding::Literal(program::Literal::Bytes(bytes))) });
                                continue;
                            }
                            _ => return None,
                        };
                        let index = bindings.cells.len();
                        bindings.cells.push(cell);
                        bindings.fields.push(Field { object, name: name.into_vec().into(),
                            value: FieldValue::Value(Binding::Cell(index)) });
                    }
                }
                object
            };
            upvalues.push(Binding::Object(object));
            stack.set_top(function_index).ok()?;
        }
        stack.set_top(function_index - 1).ok()?;
        captures.push(upvalues);
    }
    let mut visited = vec![false; tables.len()];
    while let Some(index) = pending.pop() {
        if visited[index] { continue; }
        visited[index] = true;
        let targets = setter_targets(stack, store, tables[index]).ok()?;
        let mut motion = None;
        'method: for (requested, selected) in scalar_methods::Setter::ALL.into_iter().zip(targets) {
            let Some((setter, identity)) = selected else { continue; };
            let Some(&target) = indices.get(&identity) else { continue; };
            if setter != scalar_methods::Setter::Visible {
                let ready = match motion {
                    Some((previous, ready)) if previous == target => ready,
                    _ => {
                        // Implement against Rust's canonical queue/effect owner,
                        // plus the same raw override checks as the native gate.
                        let ready = position_ready(stack, store, tables[target]).ok()?;
                        motion = Some((target, ready)); ready
                    }
                };
                if !ready { continue; }
            }
            if target != index && let Some(mt) = metatable(stack, tables[target]).ok()? {
                let method = raw(stack, mt, b"__newindex").ok()?;
                if method != Value::Nil && store.lock().unwrap().scalar_methods.identity(method)
                    != Some(scalar_methods::Method::Assign) { continue 'method; }
            }
            pending.push(target);
            let (opcode, parameters): (_, &'static [Parameter]) = match setter {
                scalar_methods::Setter::Visible => (0, &[Parameter::Receiver, Parameter::Type(Types::BOOLEAN)]),
                scalar_methods::Setter::X => (1, &[Parameter::Receiver, Parameter::Type(Types::NUMBER)]),
                scalar_methods::Setter::Y => (2, &[Parameter::Receiver, Parameter::Type(Types::NUMBER)]),
            };
            bindings.fields.push(Field { object: index as u32 + 2, name: requested.name().as_bytes().into(),
                value: FieldValue::Action(Action { tag: ((target as u32) << 2) | opcode,
                    parameters: parameters.into(), result: Some(Binding::Object(target as u32 + 2)) }) });
        }
    }
    let mut bytecode = if byte_fields.is_empty() { Vec::new() }
        else { (0..callback_count).map(|_| None).collect::<Vec<_>>() };
    let mut lower = |bindings: &program::Bindings| {
        let compiler = program::Compiler::new(bindings).ok()?;
        let mut batch = (callback_count >= 8).then(|| compiler.batch());
        let mut programs = Vec::with_capacity(callback_count);
        for (index, (callback, upvalues)) in callbacks().zip(&captures).enumerate() {
            let update_arguments;
            let arguments = if index < input_count { &bindings.arguments[..] }
                else {
                    update_arguments = [Binding::Object(updates[index - input_count].1 as u32 + 2),
                        Binding::Input { slot: 0, types: Types::NUMBER }];
                    &update_arguments[..]
                };
            let temporary;
            let bytes = if bytecode.is_empty() {
                temporary = dump(stack, *callback).ok()??; &temporary[..]
            } else {
                let bytes = &mut bytecode[index];
                if bytes.is_none() { *bytes = Some(dump(stack, *callback).ok()??); }
                &bytes.as_ref()?[..]
            };
            let program = if let Some(batch) = &mut batch { batch.compile_bytecode(bytes, arguments, upvalues) }
                else { compiler.compile_bytecode(bytes, arguments, upvalues) };
            programs.push(program.ok()?);
        }
        Some(programs)
    };
    let programs = if let Some(programs) = lower(&bindings) { programs }
        else {
            if byte_fields.is_empty() { return None; }
            for field in byte_fields {
                let cell = bindings.cells.len();
                let FieldValue::Value(Binding::Literal(program::Literal::Bytes(value))) =
                    std::mem::replace(&mut bindings.fields[field].value, FieldValue::Value(Binding::Cell(cell)))
                    else { unreachable!("captured byte literal"); };
                let index = if bindings.strings.last().is_some_and(|bytes| *bytes == value) { bindings.strings.len() - 1 }
                    else { let index = bindings.strings.len(); bindings.strings.push(value); index };
                bindings.cells.push(program::Cell::Bytes(index));
            }
            lower(&bindings)?
        };
    let mut scalar_fields = vec![(0usize, Box::<[u8]>::default()); bindings.cells.len()];
    for field in &mut bindings.fields {
        if let FieldValue::Value(Binding::Cell(index)) = &field.value {
            scalar_fields[*index] = (field.object.checked_sub(first)? as usize,
                std::mem::take(&mut field.name).into_owned().into_boxed_slice());
        }
    }
    Some(CompiledActorInputs { callbacks: Group::new(programs).ok()?, input_count,
        scalar_fields: scalar_fields.into_boxed_slice(),
        states: states.iter().copied().chain(tracked.iter().map(|(_, _, state)| *state)).collect(),
        overlay_count: states.len(), tracked: tracked.iter().map(|(target, _, _)| *target).collect() })
}
