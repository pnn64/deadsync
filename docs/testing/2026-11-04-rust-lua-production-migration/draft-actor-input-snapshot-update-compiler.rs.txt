// Draft only: the source jobs must come from the existing cached lifecycle
// plan, restricted to the authored root's Enter..Callback span. Plan roots and
// callback values remain on the caller's temporary rooted stack until lowering
// finishes. Building or executing a replacement traversal here is forbidden.
fn optional_number(stack: &mut NativeStack<'_, '_>, value: Value) -> Result<Option<f64>, Error> {
    if value == Value::Nil { return Ok(None); }
    let top = stack.top();
    stack.push(value)?;
    let number = stack.to_number(-1);
    stack.set_top(top)?;
    number.map(Some).ok_or_else(|| stack.error(b"number expected"))
}

pub(super) fn compile_source(
    stack: &mut NativeStack<'_, '_>, store: &Shared, inputs: &[Value],
    source: &[SourceJob], actors: &[TableId], states: &[SongLuaOverlayState],
    tracked: &[(SongLuaTrackedActorTarget, TableId, SongLuaOverlayState)],
) -> Option<CompiledActorInteractions> {
    let indices = actors.iter().copied().chain(tracked.iter().map(|(_, actor, _)| *actor))
        .enumerate().map(|(index, actor)| (actor, index)).collect::<FxHashMap<_, _>>();
    let mut reserved = smallvec::SmallVec::<[TableId; 8]>::new();
    let mut nodes = Vec::new();
    let mut jobs = Vec::new();
    let mut updates = Vec::new();
    for job in source {
        match *job {
            SourceJob::Enter { actor, parent, wrappers } => {
                let index = indices.get(&actor).copied();
                if index.is_none() { reserved.push(actor); }
                for key in [b"__songlua_live_delta".as_slice(), b"__songlua_recurring_update_command"] {
                    if raw(stack, actor, key).ok()? != Value::Nil { return None; }
                }
                if raw(stack, actor, b"__songlua_update_pending").ok()?.is_truthy()
                    || raw(stack, actor, b"__songlua_live_inactive").ok()?.is_truthy() { return None; }
                let rate = raw(stack, actor, b"__songlua_update_rate").ok()?;
                // Rust currently stores SetUpdateRate's canonical value in its
                // Actor, unlike the native table representation. Preserve that
                // value when no raw override exists; validate explicit values
                // with the same Lua number conversion as the native factory.
                let canonical_rate = store.lock().unwrap().actors.get(&actor)
                    .map_or(1., |actor| actor.update_rate);
                let rate = optional_number(stack, rate).ok()?.map_or(canonical_rate, |rate| rate as f32);
                let sleeping = raw(stack, actor, b"__songlua_hibernate_seconds").ok()?;
                let sleeping = optional_number(stack, sleeping).ok()?.unwrap_or(0.) as f32;
                let paused = raw(stack, actor, b"__songlua_theme_hibernating").ok()?.is_truthy();
                jobs.push(Job::Enter(nodes.len()));
                nodes.push(Node { actor: index, parent, wrappers, rate, sleeping, paused });
            }
            SourceJob::Callback { actor, slot } => {
                match raw(stack, actor, b"__songlua_update_function").ok()? {
                    Value::Nil => {},
                    function @ Value::Function(_) => {
                        // Match native Function handles: retain even callbacks
                        // read from a weak actor table throughout compilation.
                        stack.push(function).ok()?;
                        jobs.push(Job::Callback { slot, program: updates.len() });
                        updates.push((function, *indices.get(&actor)?));
                    }
                    _ => return None,
                }
            }
        }
    }
    let owner = super::compile_with_updates(stack, store, inputs, actors, states, tracked, &updates, &reserved)?;
    for job in &mut jobs {
        if let Job::Callback { program, .. } = job { *program += owner.input_count; }
    }
    let deltas = vec![Delta::default(); nodes.len()].into_boxed_slice();
    Some(CompiledActorInteractions { owner, nodes: nodes.into_boxed_slice(),
        jobs: jobs.into_boxed_slice(), deltas })
}
