#[test]
fn preparation_and_live_yoffset_keep_pause_speed_ramps_origin_and_music_rate() {
    use deadsync_rules::timing::{DelaySegment,SpeedSegment,SpeedUnit,StopSegment,TimingData,TimingSegments};
    let reference:serde_json::Value=serde_json::from_str(include_str!("../../../tests/fixtures/itgmania-actors/pause-offset.json")).unwrap();
    let mut checked=0;
    for map in reference["maps"].as_array().unwrap() {
        let name=map["name"].as_str().unwrap();
        let timing=TimingData::from_segments(0.125,0.,&TimingSegments {
            bpms:vec![(0.,120.)],
            stops:if name=="stop"{vec![StopSegment{beat:4.,duration:1.}]}else{vec![]},
            delays:if name=="delay"{vec![DelaySegment{beat:4.,duration:1.}]}else{vec![]},
            speeds:vec![SpeedSegment{beat:0.,ratio:0.5,delay:0.,unit:SpeedUnit::Beats},SpeedSegment{beat:4.,ratio:2.,delay:2.,unit:SpeedUnit::Seconds}],
            ..Default::default()
        },&[]);
        for live in [false,true] {for rate in [0.5_f32,0.75,1.,1.25,1.5,2.] {
            let mut c=context();c.live=live;c.song_music_rate=rate;c.song_timing=Some(timing.clone());c.player_timing=[Some(timing.clone()),Some(timing.clone())];
            let mut p=Pair::new(&c);
            for sample in map["samples"].as_array().unwrap() {
                let elapsed=sample["elapsed"].as_f64().unwrap()as f32;let beat=sample["beat"].as_f64().unwrap()as f32;
                let seconds=if live {f64::from(elapsed+timing.get_time_for_beat_exact(0.))}else{f64::from(elapsed)/f64::from(rate)};
                p.frame(f64::from(beat),seconds,0.);
                for player in ["nil","PS","OTHER"] {
                    let v=p.query(&format!("return ArrowEffects.GetYOffset({player},1,{})",sample["note"]));
                    assert_eq!(v.to_bits(),(sample["offset"].as_f64().unwrap()as f32).to_bits(),"{name} live={live} rate={rate} {sample}");checked+=1;
                }
            }
        }}
    }
    assert!(checked>100);
}

#[test]
fn cmod_and_reentrant_option_getters_keep_the_existing_selection_and_wave_order() {
    use deadsync_rules::timing::{TimingData,TimingSegments};
    let timing=TimingData::from_segments(0.125,0.,&TimingSegments{bpms:vec![(0.,120.)],..Default::default()},&[]);
    for live in [false,true] {
        let mut c=context();c.live=live;c.song_timing=Some(timing.clone());c.player_timing=[Some(timing.clone()),None];c.song_music_rate=1.5;
        let mut p=Pair::new(&c);p.frame(2.,1.,1.);
        for script in ["PO:CMod(300);PO:Wave(.4);PO:WavePeriod(.2)","PO:MMod(350)","PO:AMod(250)","PO:XMod(2)"] {
            p.run(script);
            for note in [-3.,0.,2.,4.,12.]{p.query(&format!("return ArrowEffects.GetYOffset(PS,1,{note})"));p.query(&format!("return ArrowEffects.GetYOffset(OTHER,1,{note})"));}
        }
        p.run(r#"LOG={};local original=PO;local count=0
            FAKE={GetPlayerNumber=function()LOG[#LOG+1]='player';return PLAYER_1 end,
            GetCurrentPlayerOptions=function()count=count+1;LOG[#LOG+1]='options'..count;collectgarbage('collect');return original end}
            ArrowEffects.GetYOffset(FAKE,1,8);assert(#LOG>=4);return table.concat(LOG,',')
        "#);
    }
}
