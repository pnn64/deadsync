cmake_minimum_required(VERSION 3.20)
project(deadlib_png_oracle C)
if(NOT DEFINED REFERENCE_ROOT)
  message(FATAL_ERROR "Set REFERENCE_ROOT to a directory containing libpng and zlib checkouts")
endif()
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${REFERENCE_ROOT}/zlib" zlib)
option(DEADLIB_ZLIB_STRICT_WINDOW "Enable zlib's advertised-window distance check" OFF)
if(DEADLIB_ZLIB_STRICT_WINDOW)
  target_compile_definitions(zlibstatic PRIVATE INFLATE_STRICT)
endif()
option(DEADLIB_ZLIB_PKZIP_WORKAROUND "Allow zlib's extended dynamic code-count definitions" OFF)
if(DEADLIB_ZLIB_PKZIP_WORKAROUND)
  target_compile_definitions(zlibstatic PRIVATE PKZIP_BUG_WORKAROUND)
endif()
add_library(ZLIB::ZLIB INTERFACE IMPORTED GLOBAL)
set_property(TARGET ZLIB::ZLIB PROPERTY INTERFACE_LINK_LIBRARIES zlibstatic)
target_include_directories(zlibstatic PUBLIC "${REFERENCE_ROOT}/zlib" "${CMAKE_CURRENT_BINARY_DIR}/zlib")
set(ZLIB_INCLUDE_DIR "${REFERENCE_ROOT}/zlib;${CMAKE_CURRENT_BINARY_DIR}/zlib")
set(ZLIB_LIBRARY zlibstatic)
set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_STATIC ON CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
set(PNG_EXECUTABLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${REFERENCE_ROOT}/libpng" libpng)
add_executable(deadlib_png_oracle oracle.c)
target_link_libraries(deadlib_png_oracle PRIVATE png_static)
add_executable(deadlib_png_encoder encode.c)
target_link_libraries(deadlib_png_encoder PRIVATE png_static)
