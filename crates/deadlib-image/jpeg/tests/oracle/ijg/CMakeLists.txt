cmake_minimum_required(VERSION 3.20)
project(deadlib_ijg_oracle C)
if(NOT JPEG_SOURCE_DIR)
  message(FATAL_ERROR "JPEG_SOURCE_DIR must point to the pinned IJG source")
endif()
set(files jaricom jcapimin jcapistd jcarith jccoefct jccolor jcdctmgr jchuff jcinit jcmainct jcmarker jcmaster jcomapi jcparam jcprepct jcsample jctrans jdapimin jdapistd jdarith jdatadst jdatasrc jdcoefct jdcolor jddctmgr jdhuff jdinput jdmainct jdmarker jdmaster jdmerge jdpostct jdsample jdtrans jerror jfdctflt jfdctfst jfdctint jidctflt jidctfst jidctint jquant1 jquant2 jutils jmemmgr jmemnobs)
set(sources)
foreach(file IN LISTS files)
  list(APPEND sources "${JPEG_SOURCE_DIR}/${file}.c")
endforeach()
add_library(ijg STATIC ${sources})
target_include_directories(ijg PUBLIC "${JPEG_SOURCE_DIR}")
add_executable(ijg-coefficients ../ijg_coefficients.c)
target_link_libraries(ijg-coefficients PRIVATE ijg)
