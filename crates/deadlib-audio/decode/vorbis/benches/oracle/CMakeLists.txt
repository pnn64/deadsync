cmake_minimum_required(VERSION 3.16)
project(deadlib_vorbis_benchmark C CXX)
if(NOT DEFINED REFERENCE_ROOT)
    message(FATAL_ERROR "Set REFERENCE_ROOT to pinned ogg/vorbis source directories")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(INSTALL_DOCS OFF CACHE BOOL "" FORCE)
set(INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(INSTALL_CMAKE_PACKAGE_MODULE OFF CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
add_subdirectory("${REFERENCE_ROOT}/ogg" ogg)
set(OGG_INCLUDE_DIR "${REFERENCE_ROOT}/ogg/include" CACHE PATH "" FORCE)
set(OGG_LIBRARY ogg CACHE STRING "" FORCE)
add_subdirectory("${REFERENCE_ROOT}/vorbis" vorbis)
option(DEADLIB_TRACK_ALLOCATIONS "Instrument only the reference libraries for heap accounting" OFF)
if(NOT DEADLIB_TRACK_ALLOCATIONS)
    add_executable(deadlib_vorbis_benchmark benchmark.cpp)
    target_include_directories(deadlib_vorbis_benchmark PRIVATE "${REFERENCE_ROOT}/vorbis/lib")
    target_compile_features(deadlib_vorbis_benchmark PRIVATE cxx_std_17)
    target_link_libraries(deadlib_vorbis_benchmark PRIVATE vorbisfile vorbis)
endif()
add_executable(deadlib_vorbis_resources resources.cpp)
target_include_directories(deadlib_vorbis_resources PRIVATE "${REFERENCE_ROOT}/vorbis/lib")
target_compile_features(deadlib_vorbis_resources PRIVATE cxx_std_17)
target_link_libraries(deadlib_vorbis_resources PRIVATE vorbisfile vorbis)
if(WIN32)
    target_link_libraries(deadlib_vorbis_resources PRIVATE psapi)
endif()
if(DEADLIB_TRACK_ALLOCATIONS)
    foreach(reference_target ogg vorbis vorbisfile)
        if(MSVC)
            target_compile_options(${reference_target} PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/alloc_profile.h")
        else()
            target_compile_options(${reference_target} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/alloc_profile.h")
        endif()
    endforeach()
    target_sources(deadlib_vorbis_resources PRIVATE alloc_profile.cpp)
    target_compile_definitions(deadlib_vorbis_resources PRIVATE DEADLIB_ALLOC_PROFILE)
endif()
