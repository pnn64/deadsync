cmake_minimum_required(VERSION 3.16)
project(deadlib_flac_oracle C)
if(NOT DEFINED REFERENCE_ROOT)
    message(FATAL_ERROR "Set REFERENCE_ROOT to the pinned libFLAC source checkout")
endif()
set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
set(WITH_OGG OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${REFERENCE_ROOT}" flac)
add_executable(deadlib_flac_oracle oracle.c)
target_link_libraries(deadlib_flac_oracle PRIVATE FLAC)
