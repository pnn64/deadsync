//! Resident compressed-sequence indexing, block decode and held-frame seeks.
use deadlib_audio_decode_wav::{
    Adpcm, AdpcmKind, AdpcmSpec, AdpcmWavl, Container, Endian, Format, Wave, WavlMuxer,
    WriteOptions,
};
use std::{hint::black_box, io::Cursor, time::Instant};
#[cfg(windows)]
fn cycles() -> u64 {
    #[link(name = "kernel32")]
    unsafe extern "system" {
        fn GetCurrentThread() -> *mut core::ffi::c_void;
        fn QueryThreadCycleTime(t: *mut core::ffi::c_void, n: *mut u64) -> i32;
    }
    let mut n = 0;
    // SAFETY: the thread pseudo-handle is valid here and n is writable during the call.
    assert_ne!(
        unsafe { QueryThreadCycleTime(GetCurrentThread(), &mut n) },
        0
    );
    n
}
#[cfg(not(windows))]
fn cycles() -> u64 {
    0
}
fn bench(name: &str, iters: usize, bytes: usize, mut work: impl FnMut()) {
    work();
    for trial in 0..3 {
        let start = Instant::now();
        let cpu = cycles();
        for _ in 0..iters {
            work();
        }
        let cpu = cycles() - cpu;
        let ns = start.elapsed().as_nanos() as f64 / iters as f64;
        println!(
            "{name}\t{trial}\t{iters}\t{ns:.3}\t{:.3}\t{:.3}",
            cpu as f64 / iters as f64,
            bytes as f64 / (1 << 20) as f64 * 1e9 / ns
        );
    }
}
fn main() {
    println!(
        "# AdpcmWavl state_bytes={}; 512 data children, four blocks per data child, three held frames after each; output bytes are decoded i16",
        std::mem::size_of::<AdpcmWavl<'_>>()
    );
    println!("operation\ttrial\titers\tns_per_op\tthread_cycles_per_op\tdecoded_MiB_per_s");
    for (name, kind, channels, block_bytes) in [
        ("ms1", AdpcmKind::Microsoft, 1, 256),
        ("ms2", AdpcmKind::Microsoft, 2, 256),
        ("ima4", AdpcmKind::Ima, 2, 256),
        ("ima3", AdpcmKind::Ima3, 2, 248),
    ] {
        let spec = AdpcmSpec {
            kind,
            channels,
            sample_rate: 8000,
            block_bytes,
        };
        let mut fmt = [0; 50];
        let len = spec.format(Endian::Little, &mut fmt).unwrap();
        let codec =
            Adpcm::from_format(Format::parse(&fmt[..len], Endian::Little).unwrap()).unwrap();
        let source = vec![0; codec.scratch_samples()];
        let mut encoded = vec![0; codec.block_bytes()];
        assert_eq!(
            codec.encode_block(&source, &mut encoded).unwrap(),
            encoded.len()
        );
        let child = encoded.repeat(4);
        let data_frames = codec.frames_per_block() as u64 * 4;
        let mut mux = WavlMuxer::new(
            Cursor::new(Vec::new()),
            &fmt[..len],
            WriteOptions {
                container: Container::Riff,
                ..WriteOptions::default()
            },
            &[],
        )
        .unwrap();
        for _ in 0..512 {
            mux.write_data(&child, data_frames).unwrap();
            mux.write_silence(3).unwrap();
        }
        let bytes = mux.finish().unwrap().into_inner();
        let wave = Wave::parse(&bytes).unwrap();
        let view = AdpcmWavl::new(&wave).unwrap();
        let mut scratch = vec![0; codec.scratch_samples()];
        let mut out = vec![1; view.frames() as usize * codec.channels()];
        assert_eq!(
            view.read_frames(0, &mut out, &mut scratch).unwrap(),
            out.len()
        );
        assert!(out.iter().all(|&n| n == 0));
        bench(&format!("{name}/index_retained_wave"), 10000, 0, || {
            black_box(AdpcmWavl::new(black_box(&wave)).unwrap());
        });
        bench(&format!("{name}/open_and_index"), 10000, 0, || {
            let wave = Wave::parse(black_box(&bytes)).unwrap();
            black_box(AdpcmWavl::new(&wave).unwrap());
        });
        let output_bytes = out.len() * 2;
        bench(&format!("{name}/full_read"), 32, output_bytes, || {
            black_box(
                view.read_frames(0, black_box(&mut out), &mut scratch)
                    .unwrap(),
            );
        });
        let mut short = [0; 62];
        let mut next = 0u64;
        bench(&format!("{name}/seek_data_31_samples"), 100000, 0, || {
            next = (next + 137) % 512;
            black_box(
                view.read_frames(
                    next * (data_frames + 3) + 17,
                    black_box(&mut short),
                    &mut scratch,
                )
                .unwrap(),
            );
        });
        bench(&format!("{name}/seek_held_frame"), 100000, 0, || {
            next = (next + 137) % 512;
            black_box(
                view.read_frames(
                    next * (data_frames + 3) + data_frames + 1,
                    black_box(&mut short[..usize::from(channels)]),
                    &mut scratch,
                )
                .unwrap(),
            );
        });
    }
}
